// Análisis de Sistemas — Microservicio Validador de Llaves
// Solución O(n) tiempo, O(1) espacio — sin estructuras de datos nativas de Java

public class BraceValidator {

    /**
     * Verifica si la secuencia de llaves es correctamente construida.
     * Cada '{' debe tener su '}' correspondiente, incluso anidados.
     *
     * @param  s  cadena de caracteres '{' y '}'
     * @return    true si la secuencia es válida, false en caso contrario
     */
    public static boolean isValid(String s) {
        int contador = 0;

        // Recorre cada carácter usando charAt()
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);

            if (c == '{') {
                contador++;
            } else if (c == '}') {
                contador--;
                // Cierre sin apertura previa → inválido de inmediato
                if (contador < 0) return false;
            }
        }

        // Válido sólo si todas las aperturas fueron cerradas
        return contador == 0;
    }

    // Alternativa con for-each (idéntica lógica)
    public static boolean isValidForEach(String s) {
        int contador = 0;
        for (char c : s.toCharArray()) {
            if      (c == '{') contador++;
            else if (c == '}') { if (--contador < 0) return false; }
        }
        return contador == 0;
    }
}
